دالة جمع_الزوجية(قائمة):
    مجموع = 0
    لكل عنصر في قائمة:
        اذا عنصر % 2 == 0:
            مجموع += عنصر
    ارجع مجموع

اطبع(جمع_الزوجية([1,2,3,4,5,6]))
# مثال برمجة كائنية
صف شخص:
    دالة __بناء__(هذا, الاسم, العمر):
        هذا.اسم = الاسم
        هذا.عمر = العمر

    دالة ترحيب(هذاي):
      اطبع("مرحبا، اسمي " + هذا.اسم + " وعمري " + str(هذا.عمر))

احمد = شخص("أحمد", 30)
احمد.ترحيب()
