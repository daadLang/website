# القوائم في لغة ض

<div dir="rtl">

## إنشاء القوائم

```daad
# قائمة فارغة
قائمة = []

# قائمة بعناصر
أرقام = [1، 2، 3، 4، 5]
أسماء = ["أحمد"، "محمد"، "سارة"]
مختلط = [1، "نص"، 3.14، صحيح]
```

## الوصول للعناصر

```daad
أرقام = [10، 20، 30، 40، 50]

اطبع(أرقام[0])    # 10 (العنصر الأول)
اطبع(أرقام[-1])   # 50 (العنصر الأخير)
اطبع(أرقام[1:3])  # [20، 30] (من 1 إلى 2)
```

## تعديل القوائم

```daad
قائمة = [1، 2، 3]

# تعديل عنصر
قائمة[0] = 10

# إضافة عنصر (append)
# قائمة.أضف(4)

# حذف عنصر
# del قائمة[0]
```

## التكرار على القوائم

```daad
فواكه = ["تفاح"، "موز"، "برتقال"]

لكل فاكهة في فواكه:
    اطبع(فاكهة)
```

## دوال القوائم

```daad
أرقام = [3، 1، 4، 1، 5، 9، 2، 6]

# الطول
اطبع(طول(أرقام))  # 8

# البحث
إذا 5 في أرقام:
    اطبع("موجود")
```

## أمثلة عملية

```daad
# حساب المجموع
أرقام = [10، 20، 30، 40]
مجموع = 0
لكل رقم في أرقام:
    مجموع += رقم
اطبع("المجموع: " + نص(مجموع))

# إيجاد الأكبر
قائمة = [3، 7، 2، 9، 1]
أكبر = قائمة[0]
لكل عدد في قائمة:
    إذا عدد > أكبر:
        أكبر = عدد
اطبع("الأكبر: " + نص(أكبر))
```
