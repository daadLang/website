# القواميس في لغة ض

<div dir="rtl">

## إنشاء القواميس

```ض
# قاموس فارغ
قاموس = {}

# قاموس بقيم
شخص = {
    "اسم": "أحمد"،
    "عمر": 25،
    "مدينة": "الرياض"
}
```

## الوصول للقيم

```ض
شخص = {"اسم": "أحمد"، "عمر": 25}

اطبع(شخص["اسم"])   # أحمد
اطبع(شخص["عمر"])   # 25
```

## تعديل القواميس

```ض
شخص = {"اسم": "أحمد"}

# تعديل قيمة
شخص["اسم"] = "محمد"

# إضافة عنصر جديد
شخص["عمر"] = 30
شخص["مدينة"] = "جدة"

# حذف عنصر
# del شخص["عمر"]
```

## التكرار على القواميس

```ض
درجات = {"رياضيات": 90، "علوم": 85، "عربي": 95}

# التكرار على المفاتيح
لكل مادة في درجات:
    اطبع(مادة + ": " + نص(درجات[مادة]))
```

## مثال: دفتر عناوين

```ض
عناوين = {
    "أحمد": "0501234567"،
    "محمد": "0559876543"،
    "سارة": "0551112233"
}

اسم = ادخل("أدخل الاسم: ")
إذا اسم في عناوين:
    اطبع("رقم " + اسم + ": " + عناوين[اسم])
وإلا:
    اطبع("الاسم غير موجود")
```
