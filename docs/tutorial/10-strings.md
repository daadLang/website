# النصوص في لغة ض

<div dir="rtl">

## إنشاء النصوص

```daad
# علامات مزدوجة
اسم = "أحمد"

# علامات مفردة
مدينة = 'الرياض'

# نص متعدد الأسطر
رسالة = """مرحباً
هذا نص متعدد
الأسطر"""
```

## دمج النصوص

```daad
الاسم_الأول = "أحمد"
الاسم_الأخير = "محمد"
الاسم_الكامل = الاسم_الأول + " " + الاسم_الأخير

# التكرار
خط = "=" * 20  # ====================
```

## الوصول للأحرف

```daad
نص = "مرحباً"
اطبع(نص[0])    # م
اطبع(نص[-1])   # ا
اطبع(نص[0:3])  # مرح
```

## دوال النصوص

```daad
نص = "مرحباً بالعالم"

# الطول
اطبع(طول(نص))

# البحث
اطبع("مرحباً" في نص)  # صحيح

# الاستبدال (إذا كانت متاحة)
# جديد = نص.استبدل("مرحباً"، "أهلاً")
```

## الأحرف الخاصة

```daad
# سطر جديد
اطبع("السطر 1
السطر 2")

# تبويب
اطبع("عمود1	عمود2")

# علامة اقتباس
اطبع("قال: "مرحباً"")

# شرطة عكسية
اطبع("المسار: C:\\مجلد")
```
