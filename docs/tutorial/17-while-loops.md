# حلقة طالما في لغة ض

<div dir="rtl">

## بناء الجملة

```ض
طالما شرط:
    # كود يتكرر طالما الشرط صحيح
```

## مثال بسيط

```ض
عداد = 0
طالما عداد < 5:
    اطبع(عداد)
    عداد += 1

# النتيجة: 0، 1، 2، 3، 4
```

## الخروج من الحلقة (اخرج)

```ض
عداد = 0
طالما صحيح:
    اطبع(عداد)
    عداد += 1
    
    إذا عداد >= 5:
        اخرج  # الخروج من الحلقة
```

## متابعة الحلقة (تابع)

```ض
عداد = 0
طالما عداد < 10:
    عداد += 1
    
    إذا عداد % 2 == 0:
        تابع  # تخطي الأرقام الزوجية
    
    اطبع(عداد)  # طباعة الأرقام الفردية فقط
```

## أمثلة عملية

### مثال 1: تخمين رقم

```ض
الرقم_السري = 7
محاولات = 0

طالما صحيح:
    تخمين = عدد(ادخل("خمن الرقم (1-10): "))
    محاولات += 1
    
    إذا تخمين == الرقم_السري:
        اطبع("صحيح! عدد المحاولات: " + نص(محاولات))
        اخرج
    وإذا تخمين < الرقم_السري:
        اطبع("الرقم أكبر")
    وإلا:
        اطبع("الرقم أصغر")
```

### مثال 2: جمع الأرقام

```ض
مجموع = 0

طالما صحيح:
    عدد = ادخل("أدخل رقماً (أو 'خروج' للإنهاء): ")
    
    إذا عدد == "خروج":
        اخرج
    
    مجموع += عدد(عدد)
    اطبع("المجموع الحالي: " + نص(مجموع))

اطبع("المجموع النهائي: " + نص(مجموع))
```
